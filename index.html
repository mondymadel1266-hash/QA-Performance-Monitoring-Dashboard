#!/usr/bin/env python3 """ Generate the `salt` and `hash` values for one dashboard account. Usage: python make_password_hash.py admin python make_password_hash.py ahmed.magdy python make_password_hash.py diaa.elsayed python make_password_hash.py yasmine.maher Then open the dashboard HTML, find the matching account inside the `ACCOUNTS` array (search for "id": "" near the top of the file) and replace its `salt` and `hash` values with the printed ones. The digest scheme must match the browser code exactly: hash = SHA-256( salt + "|" + account_id + "|" + password ) (hex) """ import getpass import hashlib import secrets import sys VALID = {"admin", "ahmed.magdy", "diaa.elsayed", "yasmine.maher"} def main() -> int: if len(sys.argv) != 2 or sys.argv[1] not in VALID: print(__doc__) return 1 account = sys.argv[1] pw = getpass.getpass(f"New password for '{account}': ") if len(pw) < 10: print("Please use at least 10 characters.") return 1 if pw != getpass.getpass("Repeat password: "): print("Passwords do not match.") return 1 salt = secrets.token_hex(16) digest = hashlib.sha256(f"{salt}|{account}|{pw}".encode("utf-8")).hexdigest() print(f'\nAccount: {account}\n"salt": "{salt}",\n"hash": "{digest}"') return 0 if __name__ == "__main__": raise SystemExit(main())
